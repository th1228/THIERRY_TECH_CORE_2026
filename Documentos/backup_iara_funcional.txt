import mediapipe as mp

def ligar_scanner_iara():
    # 1. Configura a inteligência de mapeamento
    mp_face_mesh = mp.solutions.face_mesh
    face_mesh = mp_face_mesh.FaceMesh(max_num_faces=1, refine_landmarks=True)
    mp_drawing = mp.solutions.drawing_utils

    # 2. Liga a webcam do Asus
    cap = cv2.VideoCapture(0)

    print("Iara Percept: Câmera Ativada! Pressione 'q' para sair.")

    while cap.isOpened():
        sucesso, frame = cap.read()
        if not sucesso:
            break

        # Inverte a imagem (efeito espelho) e converte para RGB
        frame = cv2.flip(frame, 1)
        rgb_frame = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)

        # Processa o rosto
        resultado = face_mesh.process(rgb_frame)

        # Desenha os pontos no rosto se encontrar um
        if resultado.multi_face_landmarks:
            for face_landmarks in resultado.multi_face_landmarks:
                mp_drawing.draw_landmarks(
                    image=frame,
                    landmark_list=face_landmarks,
                    connections=mp_face_mesh.FACEMESH_TESSELATION)

        # Mostra a janela da Iara
        cv2.imshow('Iara Percept Global - Scanner Facial', frame)

        # Aperte 'q' para fechar
        if cv2.waitKey(5) & 0xFF == ord('q'):
            break

    cap.release()
    cv2.destroyAllWindows()
